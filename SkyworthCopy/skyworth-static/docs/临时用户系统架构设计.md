## 二、Google Sheets 数据库设计

### Sheet 1: users (用户表)

| A | B | C | D | E | F | G | H | I | J |
|---|---|---|---|---|---|---|---|---|---|
| id | email | password_hash | full_name | phone | provider | provider_id | email_verified | created_at | status |

```
示例数据:
| 1 | nguyenvana@mail.com | $2a$10$... | Nguyen Van A | 0912345678 | email | | true | 2024-01-15 10:30:00 | active |
| 2 | userb@gmail.com | | Tran Thi B | 0988765432 | google | goog_123... | true | 2024-01-16 14:20:00 | active |
| 3 | userc@zalo.com | | Le Van C | 0123456789 | zalo | zalo_456... | true | 2024-01-17 09:00:00 | active |
```

**字段说明：**
| 字段 | 类型 | 说明 |
|------|------|------|
| id | 自增 | 用户唯一ID |
| email | 文本 | 登录账号（唯一） |
| password_hash | 文本 | BCrypt加密密码（第三方登录为空） |
| full_name | 文本 | 姓名 |
| phone | 文本 | 电话 |
| provider | 文本 | 登录方式：email/google/facebook/zalo |
| provider_id | 文本 | 第三方平台ID |
| email_verified | 布尔 | 邮箱是否已验证 |
| created_at | 时间戳 | 创建时间 |
| status | 文本 | pending/active/inactive |

## 三、API 端点设计

### Base URL

```
https://script.google.com/macros/s/{部署ID}/exec
```

### API 1: 用户注册

```
POST /exec

Request:
{
  "action": "user_register",
  "email": "user@example.com",
  "password": "123456",
  "fullName": "Nguyen Van A",
  "phone": "0912345678",
  "consentPrivacy": true,
  "consentMarketing": false
}

Response (成功):
{
  "success": true,
  "message": "Mã xác nhận đã gửi đến email của bạn",
  "verifyId": 1
}

Response (失败):
{
  "success": false,
  "message": "Email đã được sử dụng"
}
```

## 六、前端集成方案

### 1. API 客户端封装

```javascript
// assets/js/api-client.js
const API_BASE = 'https://script.google.com/macros/s/{部署ID}/exec';

const ApiClient = {
  // 用户注册
  register(data) {
    return this.post('user_register', data);
  },

  // 邮箱验证
  verifyEmail(email, code) {
    return this.post('verify_email', { email, code });
  },

  // 用户登录
  login(email, password) {
    return this.post('user_login', { email, password });
  },

  // 第三方登录
  socialLogin(provider, providerId, email, fullName) {
    return this.post('social_login', { provider, providerId, email, fullName });
  },

  // 重置密码请求
  resetPasswordRequest(email) {
    return this.post('reset_password_request', { email });
  },

  // 提交留言
  submitMessage(data) {
    return this.post('message_submit', data);
  },

  // 提交预约（需登录）
  submitAppointment(data) {
    return this.post('appointment_submit', data);
  },

  // 内部方法
  post(action, data) {
    return fetch(API_BASE, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ action, ...data })
    }).then(res => res.json());
  }
};
```

### 2. 用户状态管理

```javascript
// assets/js/auth.js
const Auth = {
  TOKEN_KEY: 'skyward_user_token',
  USER_KEY: 'skyward_userinfo',

  isLoggedIn() {
    return !!localStorage.getItem(this.TOKEN_KEY);
  },

  getUser() {
    const user = localStorage.getItem(this.USER_KEY);
    return user ? JSON.parse(user) : null;
  },

  login(token, user) {
    localStorage.setItem(this.TOKEN_KEY, token);
    localStorage.setItem(this.USER_KEY, JSON.stringify(user));
  },

  logout() {
    localStorage.removeItem(this.TOKEN_KEY);
    localStorage.removeItem(this.USER_KEY);
  },

  getToken() {
    return localStorage.getItem(this.TOKEN_KEY);
  }
};
```

### 3. 第三方登录处理

```javascript
// Google 登录
function onGoogleSignIn(googleUser) {
  const profile = googleUser.getBasicProfile();
  ApiClient.socialLogin(
    'google',
    profile.getId(),
    profile.getEmail(),
    profile.getName()
  ).then(res => {
    if (res.success) {
      Auth.login(res.token, res.user);
      updateUIForLoggedInUser(res.user);
    }
  });
}

// Facebook 登录
function onFacebookSignIn(response) {
  ApiClient.socialLogin(
    'facebook',
    response.userID,
    response.email,
    response.name
  ).then(res => { ... });
}

// Zalo 登录
function onZaloSignIn(user) {
  ApiClient.socialLogin(
    'zalo',
    user.id,
    user.email || user.name + '@zalo',
    user.name
  ).then(res => { ... });
}
```
